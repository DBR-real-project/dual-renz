# DualGuard 미디어 분석 파트 의존성 (Python 3.9, Windows, CPU 전용)
#
# 설치:
#   py -3.9 -m venv .venv
#   .venv\Scripts\python.exe -m pip install --upgrade pip
#   .venv\Scripts\python.exe -m pip install -r requirements.txt
#
# 주의 1: torch는 CPU 전용 휠을 쓴다. 아래 --extra-index-url이 없으면
#         pip가 CUDA 포함 휠(2GB+)을 받으려 해서 해커톤 환경에서 시간을 태운다.
# 주의 2: 시스템 PATH의 `python`은 Windows 스토어 스텁이라 동작하지 않는다.
#         반드시 .venv\Scripts\python.exe 또는 py -3.9 를 쓸 것.

--extra-index-url https://download.pytorch.org/whl/cpu

# --- 딥페이크 탐지 (HuggingFace ViT 경로, 현재 동작하는 정식 경로) ---

pretrainedmodels==0.7.4
